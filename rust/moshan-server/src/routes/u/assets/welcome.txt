const TOKEN = "__TOKEN__";
const $ = (s) => document.querySelector(s);
let ollamaOk = false;
let libDefault = "";

const FEATS = [
  {id:"pocket", name:"存图收藏",  desc:"公众号 / 小红书链接存成图集"},
  {id:"reader", name:"阅读",      desc:"订阅与稍后读"},

  {id:"ops",    name:"系统巡检",  desc:"服务、磁盘、内存"},
  {id:"media",  name:"影音室",    desc:"番号解析、封面、演员名册"},
  {id:"plan",   name:"AI 计划",   desc:"项目 → 任务 → 收件箱"}

async function api(path, body){
  const opt = body ? {method:"POST", headers:{"Content-Type":"application/json"},
                      body:JSON.stringify(body)} : {};
  const r = await fetch(path, opt);
  if(!r.ok) throw new Error(path + " -> " + r.status);
  return r.json();
}

function tag(text, kind){
  const s = document.createElement("span");
  s.className = "tag" + (kind ? " " + kind : "");
  s.textContent = text;
  return s;
}

function paintFeats(){
  const box = $("#feats");
  box.innerHTML = "";
  FEATS.forEach(f => {
    const l = document.createElement("label");
    l.className = "feat";
    l.innerHTML = '<input type="checkbox" data-f="' + f.id + '" checked>'
      + '<div><div class="fn">' + f.name + '</div><div class="fd">' + f.desc + '</div></div>';
    const cb = l.querySelector("input");
    cb.addEventListener("change", () => l.classList.toggle("off", !cb.checked));
    box.appendChild(l);
  });
}

function picked(){
  return [].slice.call(document.querySelectorAll("#feats input:checked"))
           .map(el => el.getAttribute("data-f"));
}

async function loadEnv(){
  const row = $("#envRow"); row.innerHTML = "";
  try{
    const s = await api("/welcome/status");
    const o = s.ollama || {};
    row.appendChild(tag(o.installed ? "已装 Ollama" : "未装 Ollama", o.installed ? "ok" : "warn"));
    row.appendChild(tag(o.running ? "服务运行中" : "服务未启动", o.running ? "ok" : "bad"));
    if(o.version) row.appendChild(tag(o.version));
    ollamaOk = !!o.installed && !!o.running;

    const d = s.dirs || {};
    const kv = $("#dirs"); kv.innerHTML = "";
    const add = (k,v)=>{ kv.insertAdjacentHTML("beforeend",
      '<b>'+k+'</b><span class="path">'+(v||"-")+'</span>'); };
    add("程序", d.app_dir);
    add("配置", d.config);
    add("数据", d.library_dir);

    libDefault = d.library_dir || "";
    $("#libDir").value = libDefault;
    paintLibHint();
  }catch(e){
    row.appendChild(tag("检测失败：" + e.message, "bad"));
  }
}

function paintLibHint(){
  const h = $("#libHint");
  const changed = $("#libDir").value && libDefault && $("#libDir").value !== libDefault;
  h.innerHTML = '<b>默认</b><span class="path">' + (libDefault || "-") + '</span>'
    + '<b>占用</b><span>只增不减地放你自己的文件；会话摘要那份按 300 字节/会话算，一万个会话约 3 MB</span>'
    + (changed ? '<b>改动</b><span style="color:var(--warn)">保存后重启一次中台才完全生效（旧目录不删）</span>' : '');
}

/* —— 模型来源三选 ——
   三条路最后都落到 /api/model/save：它把选择写进 config.json，同时热贴到
   正在跑的网关客户端上。只写文件是不够的 —— 进中台后客户端还拿着旧模型名，
   得等重启才生效，而用户以为已经选好了。 */
let MDL_SRC = [];

/* /api/* 要令牌，引导页地址里没有，所以单独一个带头的封装 */
async function apiT(path, body){
  const opt = body
    ? {method:"POST", headers:{"Content-Type":"application/json", "x-token":TOKEN},
       body:JSON.stringify(body)}
    : {headers:{"x-token":TOKEN}};
  const r = await fetch(path, opt);
  if(!r.ok) throw new Error(path + " -> " + r.status);
  return r.json();
}

function llmNote(msg){ $("#llmNote").textContent = msg || ""; }

function showPane(v){
  ["have","pull","api"].forEach(k => {
    $("#pane" + k.charAt(0).toUpperCase() + k.slice(1)).hidden = (k !== v);
  });
  [].slice.call($("#segSrc").children).forEach(b => {
    b.classList.toggle("on", b.dataset.v === v);
  });
}

async function saveModel(body){
  const r = await apiT("/api/model/save", body);
  if(r.ok === false) throw new Error(r.error || "保存被拒");
  const c = r.current || {};
  /* 回填到那个老输入框：最后统一 /welcome/apply 时会带 model 字段，
     不回填就把刚选的覆盖回默认的 4B 了 */
  if(c.model) $("#model").value = c.model;
  return c;
}

async function loadModels(){
  try{
    const s = await apiT("/api/model");
    MDL_SRC = s.sources || [];
    const cur = s.current || {};
    const nb = x => String(x || "").replace(/\/+$/, "");

    const sel = $("#haveSel");
    sel.innerHTML = "";
    let n = 0;
    MDL_SRC.forEach(src => {
      (src.models || []).forEach(m => {
        const o = document.createElement("option");
        o.value = [src.kind, src.base, m].join("\u0001");
        o.textContent = src.name + " · " + m + (src.kind === "gateway" ? "（可调工具）" : "");
        if(cur.base && nb(src.base) === nb(cur.base) && m === cur.model) o.selected = true;
        sel.appendChild(o); n++;
      });
    });
    if(!n){
      const o = document.createElement("option");
      o.value = ""; o.textContent = "没探到本机模型";
      sel.appendChild(o);
    }
    $("#haveHint").textContent = n
      ? "这台机上有 " + n + " 个模型，选一个点「用这个」。"
      : "没探到本机模型。切到「下载新的」拉一个，或者「接 API」。";

    const ps = $("#apiPreset");
    ps.innerHTML = "";
    MDL_SRC.forEach(src => {
      const o = document.createElement("option");
      o.value = src.id;
      o.textContent = src.name + (src.running ? "（运行中）"
                     : (src.note ? "（" + src.note + "）" : ""));
      o.dataset.base = src.base || "";
      o.dataset.kind = src.kind || "openai";
      ps.appendChild(o);
    });
    if(cur.base){
      [].slice.call(ps.options).forEach(o => {
        if(o.dataset.base && nb(o.dataset.base) === nb(cur.base)) o.selected = true;
      });
    }
    $("#apiBase").value = ps.selectedOptions[0] ? ps.selectedOptions[0].dataset.base : "";
    if(cur.model && !$("#apiModel").value) $("#apiModel").value = cur.model;

    /* 本机有就默认看本机；没有但 Ollama 在跑就给下载；都没有就停在接 API */
    showPane(n ? "have" : (ollamaOk ? "pull" : "api"));
    if(cur.model) llmNote("现在用的是 " + cur.model + "（" + (cur.base || "—") + "）");
  }catch(e){
    $("#haveHint").textContent = "读不到模型清单：" + e.message;
  }
}

async function useHave(){
  const v = $("#haveSel").value;
  if(!v){ llmNote("没有可选的模型"); return; }
  const p = v.split("\u0001");
  $("#haveBtn").disabled = true;
  try{
    const c = await saveModel({kind:p[0], base:p[1], model:p[2]});
    llmNote("已切到 " + c.model + "，进中台就能用。");
  }catch(e){ llmNote("失败：" + e.message); }
  finally{ $("#haveBtn").disabled = false; }
}

async function testApi(){
  const base = $("#apiBase").value.trim();
  const log = $("#apiLog");
  log.style.display = "block";
  if(!base){ log.textContent = "先填地址。"; return; }
  log.textContent = "探测中…";
  try{
    const r = await apiT("/api/model/test", {base, key:$("#apiKey").value.trim()});
    if(!r.ok){ log.textContent = "不通：" + (r.reason || "未知原因"); return; }
    const ms = r.models || [];
    log.textContent = "通了（" + (r.reason || "") + "），列出 " + ms.length + " 个模型：\n"
      + (ms.slice(0, 20).join("\n") || "（空）");
    if(!$("#apiModel").value.trim() && ms.length) $("#apiModel").value = ms[0];
  }catch(e){
    log.textContent = "探测失败：" + e.message;
  }
}

async function saveApi(){
  const base = $("#apiBase").value.trim(), model = $("#apiModel").value.trim();
  if(!base || !model){ llmNote("地址和模型名都要填。"); return; }
  const opt = $("#apiPreset").selectedOptions[0];
  const kind = opt ? (opt.dataset.kind || "openai") : "openai";
  $("#apiSave").disabled = true;
  try{
    const c = await saveModel({kind, base, model, key:$("#apiKey").value.trim()});
    llmNote("已保存并生效：" + c.model);
  }catch(e){ llmNote("失败：" + e.message); }
  finally{ $("#apiSave").disabled = false; }
}

async function startPull(){
  const model = $("#model").value.trim();
  if(!model) return;
  $("#pullBtn").disabled = true;
  $("#pullLog").style.display = "block";
  $("#pullLog").textContent = "";
  try{
    const r = await api("/welcome/pull", {model});
    if(!r.started){
      $("#pullLog").textContent = r.reason || "未能启动";
      $("#pullBtn").disabled = false; return;
    }
    pollPull();
  }catch(e){
    $("#pullLog").textContent = "启动失败：" + e.message;
    $("#pullBtn").disabled = false;
  }
}

async function pollPull(){
  try{
    const s = await api("/welcome/pull");
    $("#pullLog").textContent = (s.tail || []).join("\n");
    $("#pullLog").scrollTop = $("#pullLog").scrollHeight;
    if(s.running){ setTimeout(pollPull, 1200); return; }
    $("#pullBtn").disabled = false;
    if(s.ok === true){
      /* 拉完就顺手切成「用这个」，省得用户再回来选一遍 */
      try{
        await saveModel({kind:"ollama", base:"http://127.0.0.1:11434", model:s.model});
        llmNote("模型已就绪：" + s.model);
      }catch(e){
        llmNote("模型已就绪，但没写进配置：" + e.message);
      }
    }else if(s.ok === false){
      llmNote("下载未完成：" + (s.error || "未知原因"));
    }
  }catch(e){
    $("#pullBtn").disabled = false;
  }
}

async function saveThenEnter(){
  const btn = $("#doneBtn");
  btn.disabled = true; btn.textContent = "保存中…";
  const body = {
    features: picked(),
    use_llm: $("#useLlm").checked,
    model: $("#model").value.trim(),
    library_dir: $("#libDir").value.trim()
  };
  try{
    const r = await api("/welcome/apply", body);
    const st = $("#saveState");
    st.hidden = false;
    st.className = "tag " + (r.ok ? "ok" : "bad");
    st.textContent = r.ok ? ("已保存 · " + (r.moved_hint ? "重启后生效" : "可以进中台了"))
                          : ("没保存上：" + (r.error || ""));
  }catch(e){
    const st = $("#saveState");
    st.hidden = false; st.className = "tag bad";
    st.textContent = "保存失败：" + e.message;
  }
  await api("/welcome/finish", {}).catch(()=>{});
  /* 进中台，不进口袋首页 —— 口袋只是中台里的一项能力 */
  location.href = "/dash?k=" + encodeURIComponent(TOKEN);
}
